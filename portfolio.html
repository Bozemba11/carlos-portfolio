# app.py — AI IT Troubleshooting Assistant
# Step 1: Install deps → pip install flask anthropic python-dotenv

import anthropic
from flask import Flask, request, jsonify
from dotenv import load_dotenv
import os

load_dotenv()                         # Load ANTHROPIC_API_KEY from .env
app = Flask(__name__)
client = anthropic.Anthropic()       # Auto-reads ANTHROPIC_API_KEY

# The "brain" — system prompt defines how the AI behaves
SYSTEM_PROMPT = """
You are a senior IT systems engineer with 10+ years of experience.
When given an IT problem, respond with:
1. ROOT CAUSE: What is most likely causing this
2. SEVERITY: Low / Medium / High
3. STEPS: Numbered list of diagnostic/fix steps (clear, beginner-friendly)
4. TOOLS NEEDED: Any commands, software, or access required
5. PREVENTION: How to avoid this in future
Keep responses structured and practical. No jargon without explanation.
"""

@app.route('/diagnose', methods=['POST'])
def diagnose():
    data = request.get_json()
    issue = data.get('issue', '')

    if not issue:
        return jsonify({'error': 'No issue provided'}), 400

    # Call Claude — this is the key step
    message = client.messages.create(
        model="claude-opus-4-6",
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        messages=[{
            "role": "user",
            "content": f"IT Problem: {issue}"
        }]
    )

    return jsonify({
        'issue': issue,
        'diagnosis': message.content[0].text,
        'model': message.model,
        'tokens_used': message.usage.input_tokens + message.usage.output_tokens
    })

if __name__ == '__main__':
    app.run(debug=True)     # Visit http://localhost:5000
